🌐 API ENDPOINTS ADDED
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. POST /api/v1/nutrition/analyze
   Input: MultipartFile image
   Output: FoodRecognitionResponse (items + total nutrition)
   Purpose: Analyze food photo using Claude Vision

2. POST /api/v1/meals
   Input: CreateMealRequest (userId, mealType, items[], note, imageUrl)
   Output: MealResponse
   Purpose: Save meal log from recognized foods

3. GET /api/v1/meals?userId={id}&date={date}
   Output: List<MealResponse>
   Purpose: Get meals for user on specific date

4. GET /api/v1/meals/today/{userId}
   Output: DailySummaryResponse (current, target, meals, healthScore)
   Purpose: Get today's nutrition summary

5. DELETE /api/v1/meals/{id}
   Output: 204 No Content
   Purpose: Delete a meal log
